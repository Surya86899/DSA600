class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        if(nums.size()==0){return 0;}
        if(nums.size()==1){return (val==nums[0]) ? 0 : 1;}
        int i=0,j=0;
        while(j<nums.size()){
            if(nums[j]==val){j++;}
            else{
                swap(nums[i++],nums[j++]);
            }
        }
        return i;
    }
};
