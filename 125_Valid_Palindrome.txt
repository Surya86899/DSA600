class Solution {
public:
    bool isPalindrome(string s) {
        int start = 0,end = s.size()-1;
        while(start < end){
            s[start] = tolower(s[start]);
            (s[end]) = tolower(s[end]);
            if(isalnum(s[start]) && isalnum(s[end])){
                if(s[start] == s[end]){
                    start++;
                    end--;
                }
                else{
                    return false;
                }
            }
            else if(!isalnum(s[start])){
                start++;
            }
            else{
                end--;
            }
        }
        return true;
    }
};
