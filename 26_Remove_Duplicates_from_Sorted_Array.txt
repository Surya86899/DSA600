//************Brute Force Approach*****************//

class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int i = 0;
        int j = 1;
        int candidate = nums[i];
        int res=1;
        while(j<nums.size()){
            if(candidate == nums[j]){
                j++;
            }
            else{
                res++;
                i++;
                nums[i] = nums[j];
                candidate = nums[i];
                j++;
            }
        }
        i++;
        while(i<nums.size()){
            nums[i++] = 0;
        }
        return res;
    }
};


//************Optimal Prefix Suffix Solution*****************//

class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int i = 0;
        int j = 1;
        while(j<nums.size()){
            if(nums[i] == nums[j]){
                j++;
            }
            else{
                i++;
                nums[i] = nums[j];
                j++;
            }
        }
        i++;
        j = i;
        while(i<nums.size()){
            nums[i++] = 0;
        }
        return j;
    }
};
