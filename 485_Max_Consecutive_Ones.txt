class Solution {
public:
    int findMaxConsecutiveOnes(vector<int>& nums) {
        int ans = 0,i=0,j=i;
        while(j<nums.size()){
            if(nums[j]==1){
                j++;
            }
            else{
                if(ans<(j-i)){
                    ans = j-i;
                }
                j++;
                i=j;
            }
        }
        if(nums[nums.size()-1] == 1){
            if(ans<(j-i)){
                ans = j-i;
            }
        }
        return ans;
    }
};
