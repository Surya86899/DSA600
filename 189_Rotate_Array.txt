class Solution {
public:
    void swapeles(vector<int>& nums,int left,int right){
        while(left<right){
            swap(nums[left++],nums[right--]);
        }
    }
    void rotate(vector<int>& nums, int k) {
        if(nums.size()==1){return;}
        if(k>nums.size()){k%=nums.size();}

        swapeles(nums,0,nums.size()-k-1);
        swapeles(nums,nums.size()-k,nums.size()-1);
        swapeles(nums,0,nums.size()-1);
    }
};
