class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int maxprofit = 0;
        int minele = prices[0];

        for(int i = 1;i<prices.size();i++){
            minele = min(prices[i],minele);
            maxprofit = max(prices[i]-minele,maxprofit);
        }

        return maxprofit;
    }
};
