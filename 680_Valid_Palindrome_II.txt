class Solution {
public:
    bool validCheck(string& s,int start,int end){
        while(start<end){
            if(s[start]==s[end]){
                start++;
                end--;
            }
            else{
                return false;
            }
        }
        return true;
    }
    bool validPalindrome(string s) {
        int start=0,end = s.size()-1;
        bool ans = true;
        while(start<end){
            if(s[start]==s[end]){
                start++;
                end--;
            }
            else{
                break;
            }
        }
        if(start<end){
            ans = (validCheck(s,start,end-1) || validCheck(s,start+1,end));
        }
        return ans;
    }
};
